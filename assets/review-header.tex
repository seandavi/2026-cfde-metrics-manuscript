% Review-format PDF (issue #51), pulled in by _quarto-review.yml:
% continuous line numbers and double spacing. The article class
% already prints page numbers in the footer (plain page style).
\usepackage{lineno}
% Start numbering at \begin{document}: \linenumbers in the preamble
% breaks packages Quarto loads after this header (fontawesome5 under
% LuaLaTeX).
\AtBeginDocument{\linenumbers}
\usepackage{setspace}
\doublespacing
% At 12 pt the Figure 2 legend no longer fits under the full-height
% image and runs into the page number. Bound images to 0.7\textheight
% (pandoc's \pandocbounded uses \textheight) so a figure and its legend
% share one page.
\makeatletter
\renewcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{0.7\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}%
  \else\usebox{\pandoc@box}%
  \fi
}
\makeatother
